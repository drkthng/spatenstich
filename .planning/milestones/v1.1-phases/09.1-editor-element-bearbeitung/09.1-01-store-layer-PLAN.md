       - current = getZOrder(self).

    d) `sendBackward(els, id): number`:
       - current = getZOrder(self).
       - lowerOrders = others.map(getZOrder).filter(z => z < current). Wenn leer: return current.
       - return Math.max(...lowerOrders) - 1.

    e) `sendToBack(els, id): number`:
       - others = els.filter(e => e.id !== id && e.deletedAt === null).
       - min = others reduce zu Math.min(getZOrder(e)) oder 0 wenn leer.
       - return min - 1.
