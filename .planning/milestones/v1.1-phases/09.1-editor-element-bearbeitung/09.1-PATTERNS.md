```typescript
const SNAP_DEG = 15;
export function snapRotation(deg: number, freeRotation: boolean): number {
  const normalized = ((deg % 360) + 360) % 360;
  if (freeRotation) return normalized;
  return Math.round(normalized / SNAP_DEG) * SNAP_DEG % 360;
}
```
